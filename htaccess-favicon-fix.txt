# === ДОБАВИТЬ В .htaccess ПЕРЕД # BEGIN WordPress ===
# Favicon/static bypass + MIME types
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteRule ^(favicon\.svg|site\.webmanifest|icon-120\.png|apple-touch-icon\.png|icon-192\.svg|icon-512\.svg)$ - [L]
</IfModule>
<IfModule mod_mime.c>
AddType image/svg+xml .svg
AddType application/manifest+json .webmanifest
</IfModule>
